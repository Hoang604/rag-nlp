"use client";
import { State, postToSpreadSheet } from "@/src/app/actions";
import { formSchema } from "@/src/app/validation";
import { zodResolver } from "@hookform/resolvers/zod";
import { useRouter } from "next/navigation";
import React, { useEffect } from "react";
import { useFormState } from "react-dom";
import { FieldPath, FormProvider, useForm } from "react-hook-form";
import * as z from "zod";
import { FormContent } from "./components/FormContent";

export type Schema = z.infer<typeof formSchema>;

export const HeroForm: React.FC = () => {
  const [state, formAction] = useFormState<State, FormData>(
    postToSpreadSheet,
    null
  );

  const router = useRouter();

  const formMethods = useForm<Schema>({
    mode: "all",
    resolver: zodResolver(formSchema),
    defaultValues: {
      email: "",
    },
  });

  const { setError, reset } = formMethods;

  useEffect(() => {
    if (!state) return;

    if (state.status === "error") {
      console.log(state.errors);
      state.errors?.forEach((error) => {
        setError(error.path as FieldPath<Schema>, {
          message: error.message,
        });
      });
    }

    if (state.status === "success") {
      alert("Success!");
      reset();
      router.push("/success");
    }
  }, [reset, router, setError, state]);

  return (
    <FormProvider {...formMethods}>
      <form action={formAction}>
        <FormContent />
      </form>
    </FormProvider>
  );
};
