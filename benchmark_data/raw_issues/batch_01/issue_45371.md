### To Reproduce

```js
/*
The following code will reproduce the issue in any project with turbopack enabled
*/

import React from 'react';
import { cookies } from 'next/headers';
import { redirect } from 'next/navigation';

const Home = () => {

    const nextCookies = cookies();
    const auth = nextCookies.get('ccauth');

    if (!auth) {
    redirect('/login');
  }

    return (
        <div>page</div>
    );
};

export default Home;
```

App breaks when trying to access next/header cookie info on server component
